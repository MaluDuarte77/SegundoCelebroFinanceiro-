    function formatCurrency(val) {
      return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(val);
    }

    function formatDate(dateStr) {
      if (!dateStr) return '';
      const [year, month, day] = dateStr.split('-');
      return `${day}/${month}/${year}`;
    }

    function switchTab(tabId) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
      document.querySelectorAll('.nav-tab').forEach(el => {
        el.classList.remove('bg-monotonal-800', 'text-white', 'border', 'border-monotonal-700');
        el.classList.add('text-monotonal-400');
      });

      const activeContent = document.getElementById(`tab-${tabId}`);
      const activeBtn = document.getElementById(`tab-btn-${tabId}`);
      if (activeContent && activeBtn) {
        activeContent.classList.remove('hidden');
        activeBtn.classList.add('bg-monotonal-800', 'text-white', 'border', 'border-monotonal-700');
        activeBtn.classList.remove('text-monotonal-400');
      }
    }

    function calculateMetrics() {
      let income = 0, expenses = 0;
      appState.transactions.forEach(t => {
        if (t.type === 'income') income += Number(t.amount);
        else if (t.type === 'expense') expenses += Number(t.amount);
      });
      const balance = income - expenses;
      const savingsRate = income > 0 ? Math.max(0, ((income - expenses) / income) * 100) : 0;
      return { income, expenses, balance, savingsRate };
    }

    // HANDLERS & ACTIONS
    function handleAddTransaction(e) {
      e.preventDefault();
      const desc = document.getElementById('tx-description').value;
      const amount = parseFloat(document.getElementById('tx-amount').value);
      const type = document.getElementById('tx-type').value;
      const category = document.getElementById('tx-category').value;
      const date = document.getElementById('tx-date').value;

      if (!desc || isNaN(amount) || !date) return;

      appState.transactions.push({ id: Date.now().toString(), description: desc, amount, type, category, date });
      saveState();
      document.getElementById('form-transaction').reset();
      document.getElementById('tx-date').valueAsDate = new Date();
    }

    function deleteTransaction(id) {
      appState.transactions = appState.transactions.filter(t => t.id !== id);
      saveState();
    }

    function openAddNoteModal() {
      document.getElementById('modal-note-title').textContent = 'Criar Nova Nota';
      document.getElementById('form-note').reset();
      document.getElementById('note-id').value = '';
      document.getElementById('modal-note').classList.remove('hidden');
    }

    function editNote(id) {
      const note = appState.notes.find(n => n.id === id);
      if (!note) return;
      document.getElementById('modal-note-title').textContent = 'Editar Nota';
      document.getElementById('note-id').value = note.id;
      document.getElementById('note-title-input').value = note.title;
      document.getElementById('note-category').value = note.category;
      document.getElementById('note-tags').value = note.tags ? note.tags.join(', ') : '';
      document.getElementById('note-content').value = note.content;
      document.getElementById('note-fav').checked = !!note.favorite;
      document.getElementById('modal-note').classList.remove('hidden');
    }

    function handleSaveNote(e) {
      e.preventDefault();
      const id = document.getElementById('note-id').value;
      const title = document.getElementById('note-title-input').value;
      const category = document.getElementById('note-category').value;
      const tagsRaw = document.getElementById('note-tags').value;
      const content = document.getElementById('note-content').value;
      const favorite = document.getElementById('note-fav').checked;

      const tags = tagsRaw.split(',').map(t => t.trim()).filter(t => t.length > 0);

      if (id) {
        const note = appState.notes.find(n => n.id === id);
        if (note) {
          note.title = title;
          note.category = category;
          note.tags = tags;
          note.content = content;
          note.favorite = favorite;
        }
      } else {
        appState.notes.push({
          id: Date.now().toString(), title, category, tags, content, favorite,
          date: new Date().toISOString().split('T')[0]
        });
      }
      saveState();
      closeModal('modal-note');
    }

    function deleteNote(id) {
      appState.notes = appState.notes.filter(n => n.id !== id);
      saveState();
    }

    function toggleFavNote(id) {
      const note = appState.notes.find(n => n.id === id);
      if (note) {
        note.favorite = !note.favorite;
        saveState();
      }
    }

    function toggleFavFilter() {
      appState.filterOnlyFavs = !appState.filterOnlyFavs;
      document.getElementById('btn-toggle-fav-filter').classList.toggle('text-yellow-400', appState.filterOnlyFavs);
      renderBrainTab();
    }

    function openAddGoalModal() {
      document.getElementById('form-goal').reset();
      document.getElementById('modal-goal').classList.remove('hidden');
    }

    function handleSaveGoal(e) {
      e.preventDefault();
      const title = document.getElementById('goal-title').value;
      const current = parseFloat(document.getElementById('goal-current').value) || 0;
      const target = parseFloat(document.getElementById('goal-target').value);
      if (!title || isNaN(target)) return;
      appState.goals.push({ id: Date.now().toString(), title, current, target });
      saveState();
      closeModal('modal-goal');
    }

    function updateGoalProgress(id) {
      const goal = appState.goals.find(g => g.id === id);
      if (!goal) return;
      const newVal = prompt(`Atualizar valor de "${goal.title}":`, goal.current);
      if (newVal !== null && !isNaN(parseFloat(newVal))) {
        goal.current = parseFloat(newVal);
        saveState();
      }
    }

    function deleteGoal(id) {
      appState.goals = appState.goals.filter(g => g.id !== id);
      saveState();
    }

    function openAddHabitModal() {
      document.getElementById('form-habit').reset();
      document.getElementById('modal-habit').classList.remove('hidden');
    }

    function handleSaveHabit(e) {
      e.preventDefault();
      const title = document.getElementById('habit-title').value;
      if (!title) return;
      appState.habits.push({ id: Date.now().toString(), title, completed: false });
      saveState();
      closeModal('modal-habit');
    }

    function toggleHabit(id) {
      const habit = appState.habits.find(h => h.id === id);
      if (habit) {
        habit.completed = !habit.completed;
        saveState();
      }
    }

    function deleteHabit(id) {
      appState.habits = appState.habits.filter(h => h.id !== id);
      saveState();
    }

    function resetHabitsDaily() {
      appState.habits.forEach(h => h.completed = false);
      saveState();
    }

    function closeModal(id) {
      document.getElementById(id).classList.add('hidden');
    }

    function exportData() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState, null, 2));
      const a = document.createElement('a');
      a.setAttribute("href", dataStr);
      a.setAttribute("download", `segundo_cerebro_backup_${new Date().toISOString().split('T')[0]}.json`);
      document.body.appendChild(a);
      a.click();
      a.remove();
    }

    function importData(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const imported = JSON.parse(e.target.result);
          if (imported.transactions && imported.notes) {
            appState = imported;
            saveState();
            alert('Dados importados com sucesso!');
          }
        } catch (err) { alert('Ficheiro inválido.'); }
      };
      reader.readAsText(file);
    }

    function resetData() {
      if (confirm('Deseja redefinir os dados para o padrão?')) {
        appState = JSON.parse(JSON.stringify(defaultData));
        saveState();
      }
    }

    window.onload = function() {
      loadState();
      const dateInput = document.getElementById('tx-date');
      if (dateInput) dateInput.valueAsDate = new Date();
      renderApp();
    };
